### ci.yml 'Build aggregate timing summary' step, extracted via YAML parse and executed locally
--- step script ---
set -eu
mkdir -p "$RUNNER_TEMP/fm-test" "$RUNNER_TEMP/fm-test-aggregate"
# The download directory is passed whole: the aggregator resolves
# every fm-test-timing-*.json beneath it, including the Herdr lane's
# file nested inside its diagnostics bundle, validates each lane's
# identity, and refuses duplicates. A lane with no artifact (a
# cancelled job, a failed upload) fails this step by name.
rc=0
lanes=$(bin/fm-test-run.sh --list-ci-lanes) || rc=$?
if [ "$rc" -ne 0 ] || [ -z "$lanes" ]; then
  echo "::error::could not build the expected-lane manifest from bin/fm-test-run.sh --list-ci-lanes (exit $rc)"
  exit 1
fi
expect=()
while IFS= read -r lane; do
  [ -n "$lane" ] || continue
  expect+=(--expect-lane "$lane")
done <<< "$lanes"
[ "${#expect[@]}" -gt 0 ] || { echo "::error::the expected-lane manifest is empty"; exit 1; }
bin/fm-test-run.sh --aggregate-json \
  "$RUNNER_TEMP/fm-test/fm-test-timing-aggregate.json" \
  "${expect[@]}" "$RUNNER_TEMP/fm-test-aggregate" || rc=$?
if [ "$rc" -ne 0 ]; then
  echo "::error::the timing aggregate is INCOMPLETE (exit 1, see the FM_TEST_AGGREGATE lines above) or was refused (exit 2, see the fm-test-run line above); got exit $rc"
  exit "$rc"
fi
